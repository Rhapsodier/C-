# C++虚函数实例讲解

## 什么是虚函数

虚函数是C++中实现运行时多态的关键机制。通过在基类中声明虚函数，派生类可以重写这些函数，实现不同的行为。当通过基类指针或引用调用虚函数时，会根据实际对象的类型调用相应的函数版本。

## 虚函数的基本语法

```cpp
class Base {
public:
    virtual void functionName() {
        // 基类实现
    }
    
    virtual ~Base() {} // 虚析构函数
};
```

## 虚函数实例

### 基本虚函数示例

```cpp
#include <iostream>
using namespace std;

class Animal {
public:
    virtual void makeSound() {
        cout << "动物发出声音" << endl;
    }
    
    virtual ~Animal() {}
};

class Dog : public Animal {
public:
    void makeSound() override {
        cout << "汪汪汪！" << endl;
    }
};

class Cat : public Animal {
public:
    void makeSound() override {
        cout << "喵喵喵！" << endl;
    }
};

int main() {
    Animal* animals[3];
    animals[0] = new Animal();
    animals[1] = new Dog();
    animals[2] = new Cat();
    
    // 通过虚函数实现运行时多态
    for (int i = 0; i < 3; i++) {
        animals[i]->makeSound();
    }
    
    // 清理内存
    for (int i = 0; i < 3; i++) {
        delete animals[i];
    }
    
    return 0;
}
```

### 虚函数与面积计算

```cpp
#include <iostream>
using namespace std;

class Shape {
public:
    virtual double area() {
        cout << "基类面积计算" << endl;
        return 0.0;
    }
    
    virtual ~Shape() {}
};

class Circle : public Shape {
private:
    double radius;
    
public:
    Circle(double r) : radius(r) {}
    
    double area() override {
        cout << "圆形面积计算: ";
        return 3.14159 * radius * radius;
    }
};

class Rectangle : public Shape {
private:
    double width, height;
    
public:
    Rectangle(double w, double h) : width(w), height(h) {}
    
    double area() override {
        cout << "矩形面积计算: ";
        return width * height;
    }
};

int main() {
    Shape* shapes[3];
    shapes[0] = new Shape();
    shapes[1] = new Circle(5.0);
    shapes[2] = new Rectangle(4.0, 6.0);
    
    // 使用虚函数实现多态
    for (int i = 0; i < 3; i++) {
        cout << "面积: " << shapes[i]->area() << endl;
    }
    
    // 清理内存
    for (int i = 0; i < 3; i++) {
        delete shapes[i];
    }
    
    return 0;
}
```

## 虚函数的工作原理

### 虚函数表（vtable）

```cpp
#include <iostream>
using namespace std;

class Base {
public:
    virtual void func1() {
        cout << "Base::func1()" << endl;
    }
    
    virtual void func2() {
        cout << "Base::func2()" << endl;
    }
    
    virtual ~Base() {}
};

class Derived : public Base {
public:
    void func1() override {
        cout << "Derived::func1()" << endl;
    }
    
    void func2() override {
        cout << "Derived::func2()" << endl;
    }
};

int main() {
    Base* ptr = new Derived();
    
    // 调用虚函数，实际调用Derived类的版本
    ptr->func1();
    ptr->func2();
    
    delete ptr;
    return 0;
}
```

## 虚函数的注意事项

### 1. 构造函数不能是虚函数

```cpp
class Base {
public:
    // 错误：构造函数不能是虚函数
    // virtual Base() {}
    
    Base() {
        cout << "Base构造函数" << endl;
    }
    
    virtual ~Base() {}
};
```

### 2. 析构函数应该是虚函数

```cpp
class Base {
public:
    virtual ~Base() {
        cout << "Base析构函数" << endl;
    }
};

class Derived : public Base {
public:
    ~Derived() {
        cout << "Derived析构函数" << endl;
    }
};

int main() {
    Base* ptr = new Derived();
    delete ptr;  // 正确调用Derived的析构函数
    
    return 0;
}
```

### 3. 虚函数的重写规则

```cpp
class Base {
public:
    virtual void func(int x) {
        cout << "Base::func(" << x << ")" << endl;
    }
    
    virtual ~Base() {}
};

class Derived : public Base {
public:
    // 正确：完全匹配的参数列表
    void func(int x) override {
        cout << "Derived::func(" << x << ")" << endl;
    }
    
    // 错误：参数不匹配，不是重写
    // void func(double x) override { }
};
```

## 纯虚函数

```cpp
class AbstractShape {
public:
    // 纯虚函数，使类成为抽象类
    virtual double area() = 0;
    
    virtual ~AbstractShape() {}
};

// 派生类必须实现纯虚函数
class Triangle : public AbstractShape {
private:
    double base, height;
    
public:
    Triangle(double b, double h) : base(b), height(h) {}
    
    double area() override {
        return 0.5 * base * height;
    }
};
```

## 总结

虚函数是C++实现运行时多态的核心机制：

1. **基本语法**：使用`virtual`关键字声明
2. **工作原理**：通过虚函数表实现动态绑定
3. **注意事项**：
   - 构造函数不能是虚函数
   - 析构函数应该是虚函数
   - 重写时参数列表必须完全匹配
4. **应用场景**：需要根据实际对象类型调用不同函数版本的场景
