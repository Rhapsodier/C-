# C++纯虚函数与抽象类总结

## 什么是纯虚函数

纯虚函数是在基类中声明的虚函数，它没有实现，只是为派生类提供一个接口规范。纯虚函数通过在函数声明后添加 `= 0` 来标识。

## 纯虚函数的语法

```cpp
class AbstractClass {
public:
    // 纯虚函数声明
    virtual void pureFunction() = 0;
    
    // 虚析构函数
    virtual ~AbstractClass() = default;
};
```

## 什么是抽象类

包含纯虚函数的类称为抽象类。抽象类不能被实例化，只能作为基类被继承。派生类必须实现所有的纯虚函数，否则派生类也会成为抽象类。

## 纯虚函数的基本示例

### 简单的抽象类

```cpp
#include <iostream>
using namespace std;

class Shape {
public:
    // 纯虚函数
    virtual double area() = 0;
    virtual double perimeter() = 0;
    
    // 虚析构函数
    virtual ~Shape() = default;
    
    // 普通成员函数
    void displayInfo() {
        cout << "面积: " << area() << endl;
        cout << "周长: " << perimeter() << endl;
    }
};

class Circle : public Shape {
private:
    double radius;
    
public:
    Circle(double r) : radius(r) {}
    
    // 实现纯虚函数
    double area() override {
        return 3.14159 * radius * radius;
    }
    
    double perimeter() override {
        return 2 * 3.14159 * radius;
    }
};

class Rectangle : public Shape {
private:
    double width, height;
    
public:
    Rectangle(double w, double h) : width(w), height(h) {}
    
    // 实现纯虚函数
    double area() override {
        return width * height;
    }
    
    double perimeter() override {
        return 2 * (width + height);
    }
};

int main() {
    // Shape s;  // 错误：不能创建抽象类对象
    
    Shape* shapes[2];
    shapes[0] = new Circle(5.0);
    shapes[1] = new Rectangle(4.0, 6.0);
    
    for (int i = 0; i < 2; i++) {
        shapes[i]->displayInfo();
        cout << endl;
    }
    
    // 清理内存
    for (int i = 0; i < 2; i++) {
        delete shapes[i];
    }
    
    return 0;
}
```

## 纯虚函数的实现

### 纯虚函数可以有默认实现

```cpp
#include <iostream>
using namespace std;

class Animal {
public:
    // 纯虚函数，但有默认实现
    virtual void makeSound() = 0;
    
    virtual ~Animal() = default;
};

// 提供默认实现
void Animal::makeSound() {
    cout << "动物发出声音" << endl;
}

class Dog : public Animal {
public:
    // 使用默认实现
    void makeSound() override {
        Animal::makeSound();  // 调用基类的默认实现
        cout << "汪汪汪！" << endl;
    }
};

class Cat : public Animal {
public:
    // 完全重写
    void makeSound() override {
        cout << "喵喵喵！" << endl;
    }
};

int main() {
    Animal* animals[2];
    animals[0] = new Dog();
    animals[1] = new Cat();
    
    for (int i = 0; i < 2; i++) {
        animals[i]->makeSound();
    }
    
    for (int i = 0; i < 2; i++) {
        delete animals[i];
    }
    
    return 0;
}
```

## 抽象类的设计模式

### 模板方法模式

```cpp
#include <iostream>
using namespace std;

class Game {
public:
    // 模板方法
    void play() {
        initialize();
        startPlay();
        endPlay();
    }
    
    virtual ~Game() = default;
    
protected:
    // 纯虚函数，子类必须实现
    virtual void initialize() = 0;
    virtual void startPlay() = 0;
    virtual void endPlay() = 0;
};

class Cricket : public Game {
protected:
    void initialize() override {
        cout << "板球游戏初始化！" << endl;
    }
    
    void startPlay() override {
        cout << "开始板球游戏！" << endl;
    }
    
    void endPlay() override {
        cout << "板球游戏结束！" << endl;
    }
};

class Football : public Game {
protected:
    void initialize() override {
        cout << "足球游戏初始化！" << endl;
    }
    
    void startPlay() override {
        cout << "开始足球游戏！" << endl;
    }
    
    void endPlay() override {
        cout << "足球游戏结束！" << endl;
    }
};

int main() {
    Game* games[2];
    games[0] = new Cricket();
    games[1] = new Football();
    
    for (int i = 0; i < 2; i++) {
        games[i]->play();
        cout << endl;
    }
    
    for (int i = 0; i < 2; i++) {
        delete games[i];
    }
    
    return 0;
}
```

## 纯虚析构函数

### 纯虚析构函数必须提供实现

```cpp
#include <iostream>
using namespace std;

class AbstractBase {
public:
    AbstractBase() {
        cout << "AbstractBase构造函数" << endl;
    }
    
    // 纯虚析构函数
    virtual ~AbstractBase() = 0;
    
    // 纯虚函数
    virtual void pureFunc() = 0;
};

// 纯虚析构函数必须提供实现
AbstractBase::~AbstractBase() {
    cout << "AbstractBase析构函数" << endl;
}

class Concrete : public AbstractBase {
public:
    Concrete() {
        cout << "Concrete构造函数" << endl;
    }
    
    ~Concrete() override {
        cout << "Concrete析构函数" << endl;
    }
    
    void pureFunc() override {
        cout << "Concrete::pureFunc()" << endl;
    }
};

int main() {
    AbstractBase* ptr = new Concrete();
    ptr->pureFunc();
    delete ptr;
    
    return 0;
}
```

## 抽象类的应用场景

### 1. 接口定义

```cpp
class DatabaseInterface {
public:
    virtual void connect() = 0;
    virtual void disconnect() = 0;
    virtual void execute(const string& query) = 0;
    virtual ~DatabaseInterface() = default;
};

class MySQLDatabase : public DatabaseInterface {
public:
    void connect() override {
        cout << "连接MySQL数据库" << endl;
    }
    
    void disconnect() override {
        cout << "断开MySQL数据库连接" << endl;
    }
    
    void execute(const string& query) override {
        cout << "执行MySQL查询: " << query << endl;
    }
};
```

### 2. 策略模式

```cpp
class PaymentStrategy {
public:
    virtual void pay(double amount) = 0;
    virtual ~PaymentStrategy() = default;
};

class CreditCardPayment : public PaymentStrategy {
public:
    void pay(double amount) override {
        cout << "使用信用卡支付: $" << amount << endl;
    }
};

class PayPalPayment : public PaymentStrategy {
public:
    void pay(double amount) override {
        cout << "使用PayPal支付: $" << amount << endl;
    }
};

class ShoppingCart {
private:
    PaymentStrategy* paymentStrategy;
    
public:
    void setPaymentStrategy(PaymentStrategy* strategy) {
        paymentStrategy = strategy;
    }
    
    void checkout(double amount) {
        if (paymentStrategy) {
            paymentStrategy->pay(amount);
        }
    }
};
```

## 抽象类的注意事项

### 1. 不能创建抽象类对象

```cpp
class AbstractClass {
public:
    virtual void func() = 0;
    virtual ~AbstractClass() = default;
};

int main() {
    // AbstractClass obj;  // 编译错误
    // AbstractClass* ptr = new AbstractClass();  // 编译错误
    
    return 0;
}
```

### 2. 派生类必须实现所有纯虚函数

```cpp
class AbstractClass {
public:
    virtual void func1() = 0;
    virtual void func2() = 0;
    virtual ~AbstractClass() = default;
};

class ConcreteClass : public AbstractClass {
public:
    void func1() override {
        cout << "func1实现" << endl;
    }
    
    // 错误：没有实现func2，ConcreteClass仍然是抽象类
    // void func2() override { }
};

int main() {
    // ConcreteClass obj;  // 编译错误，仍然是抽象类
    
    return 0;
}
```

### 3. 构造函数和析构函数

```cpp
class AbstractClass {
public:
    // 构造函数不能是虚函数
    AbstractClass() {
        cout << "抽象类构造函数" << endl;
    }
    
    // 析构函数应该是虚函数
    virtual ~AbstractClass() = 0;
};

AbstractClass::~AbstractClass() {
    cout << "抽象类析构函数" << endl;
}
```

## 抽象类与接口的区别

### C++中的接口实现

```cpp

    virtual void print() = 0;

class Document : public Printable {
public:
    void print() override {
        cout << "打印文档" << endl;
    }
};

class Image : public Printable {
public:
    void print() override {

    }
};
```

## 总结

纯虚函数和抽象类是C++面向对象设计的重要概念：

### 特点
1. **纯虚函数**：使用 `= 0` 声明，没有实现
2. **抽象类**：包含纯虚函数的类，不能实例化
3. **强制实现**：派生类必须实现所有纯虚函数

### 应用场景
1. **接口定义**：定义类的行为规范
2. **设计模式**：模板方法、策略等模式
3. **框架设计**：提供扩展点

### 注意事项
1. 抽象类不能创建对象
2. 纯虚析构函数必须提供实现
3. 派生类必须实现所有纯虚函数
4. 构造函数不能是虚函数

抽象类是C++实现多态性和接口设计的重要工具，合理使用可以提高代码的可扩展性和维护性。
